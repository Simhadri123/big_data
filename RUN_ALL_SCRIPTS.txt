# Big Data Workspace — Run All Scripts
#
# 1) Clone the repository
# 2) Build and start the Docker container (workspace mounted at /workspace)
# 3) Execute every run.sh via the container wrapper
# 4) (Optional) Stop the container when done

# --- 1) Clone ---
 git clone https://github.com/Simhadri123/big_data.git
 cd big_data

# --- 2) Build & start container ---
 ./start-docker.sh
