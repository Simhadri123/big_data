# Big Data Workspace — Run All Scripts
#
# 1) Clone the repository
# 2) Build and start the Docker container (workspace mounted at /workspace)
# 3) Execute every run.sh via the container wrapper
# 4) (Optional) Stop the container when done

# --- 1) Clone ---
 git clone https://github.com/Simhadri123/big_data.git
 cd big_data

# --- 2) Build & start container ---
 ./start-docker.sh

# --- 3) Run all run.sh scripts ---
 # Replicated Join (spatial)
 docker-compose exec hadoop-workspace /run.sh /workspace/RepJoin/run.sh

 # Sort-based Join (no sort)
 docker-compose exec hadoop-workspace /run.sh /workspace/join_no_sort/run.sh

 # SGPA/CGPA MapReduce join
 docker-compose exec hadoop-workspace /run.sh /workspace/map_reduce_join/run.sh

 # Triangle counting — no partition
 docker-compose exec hadoop-workspace /run.sh /workspace/counting_triangles_no_partition/run.sh

 # Triangle counting — with partition
 docker-compose exec hadoop-workspace /run.sh /workspace/counting_triangles_partition/run.sh

 # Multi-way join
 docker-compose exec hadoop-workspace /run.sh /workspace/multi_way_join/run.sh

 # WordCount (Data Algorithms book - chapB01)
 docker-compose exec hadoop-workspace /run.sh /workspace/data-algorithms-book/src/main/java/org/dataalgorithms/chapB01/wordcount/mapreduce/run.sh

# --- 4) (Optional) Stop container ---
 docker-compose down
